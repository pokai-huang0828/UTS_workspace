# 偵測並設定運算裝置 (如果有可用的 GPU 就使用 GPU，否則使用 CPU)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"使用裝置: {device}")

# 定義神經網路結構 (與實踐課相同的結構) - 這裡使用了兩層卷積層和兩層全連接層的簡單 CNN 結構
class Net(nn.Module):
    def __init__(self):
        super(Net, self).__init__()
        self.conv1 = nn.Conv2d(1, 32, kernel_size=3, padding=1)
        self.conv2 = nn.Conv2d(32, 64, kernel_size=3, padding=1)
        self.pool = nn.MaxPool2d(2, 2)
        self.fc1 = nn.Linear(64 * 7 * 7, 128)
        self.fc2 = nn.Linear(128, 10)

    def forward(self, x):
        x = self.pool(F.relu(self.conv1(x)))
        x = self.pool(F.relu(self.conv2(x)))
        x = x.view(-1, 64 * 7 * 7)
        x = F.relu(self.fc1(x))
        x = self.fc2(x)
        return x

model = Net().to(device)
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# 開始訓練模型 - 在每個訓練輪數中，模型會進行前向傳播、計算損失、反向傳播和參數更新。訓練過程中會輸出每個輪數的平均損失。 這有助於觀察模型的學習進度和收斂情況。
print("開始訓練模型...")
for epoch in range(epochs):
    model.train()
    running_loss = 0.0
    for images, labels in train_loader:
        images, labels = images.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
    
    print(f"Epoch [{epoch+1}/{epochs}], Loss: {running_loss/len(train_loader):.4f}")

生成对抗样本¶

要生成对抗图像，需要构造与原图相同形状（32×32）的扰动噪声。如实践课程所述，噪声过大将使对抗样本不能被人眼识别，而噪声过小可能无法误导模型。
请谨慎缩放噪声幅度，目标是在尽可能小的噪声下成功误导模型。

然后将生成的噪声分别叠加到五张原始图像上，并输出这 5 张带噪声图像。注意，只需要为前五张图片生成对抗样本。

请展示 5 张对抗图像、它们的原始标签以及模型给出的预测标签；并将 5 张原图与 5 张对抗图像输出进行对比展示。

# 確保模型在評估模式
model.eval()

# 產生一個與 FashionMNIST 圖片相同大小 (28x28) 的隨機雜訊矩陣
noiseMatrix = np.random.random((28, 28)) 

# 測試參數：數字越大 = 雜訊越小 = 圖片越清晰，但越難欺騙模型
noise_scale = 1.9
print(f"----- 目前測試的雜訊縮放比例為: 1/{noise_scale} -----\n")

fig, axes = plt.subplots(2, 5, figsize=(15, 6))
fig.suptitle(f"Random Noise Attack (Noise scaled down to 1/{noise_scale})", fontsize=16)

# 準備空陣列來記錄每一輪的結果，避免變數被覆寫
true_labels_list = []
adv_preds_list = []

# 針對前 5 張圖片進行測試
for i in range(5):
    # 取得原始圖片與標籤
    img_tensor, true_label = train_dataset[i]
    img_numpy = img_tensor.squeeze().numpy()
    
    # 記錄真實標籤
    true_labels_list.append(true_label)
    
    # 將隨機雜訊疊加上去，並使用 np.clip 確保像素值不超過 1.0 (避免圖片失真)
    noisy_img_numpy = np.clip(img_numpy + (noiseMatrix / noise_scale), 0.0, 1.0)
    
    # 將 NumPy 陣列轉回 Tensor，並增加 Batch 與 Channel 維度，送入裝置
    noisy_images = torch.tensor(noisy_img_numpy, dtype=torch.float32)
    noisy_images = noisy_images.unsqueeze(0).unsqueeze(0).to(device)
    
    # 讓模型預測這張充滿雜訊的圖片
    with torch.no_grad():
        outputs = model(noisy_images)
        _, predicted = torch.max(outputs.data, 1)
    
    pred_label = predicted.item()
    
    # 記錄預測標籤
    adv_preds_list.append(pred_label)
    
    # --- 繪圖區 ---
    # 上排：原始圖片 (固定顏色範圍 0~1)
    axes[0, i].imshow(img_numpy, cmap='gray', vmin=0, vmax=1)
    axes[0, i].set_title(f"Orig: {classes[true_label]}")
    axes[0, i].axis('off')
    
    # 下排：加入隨機雜訊後的圖片 (固定顏色範圍 0~1)
    axes[1, i].imshow(noisy_img_numpy, cmap='gray', vmin=0, vmax=1)
    
    # 判斷是否成功欺騙
    success = true_label != pred_label
    mark = "V" if success else "X"
    color = "red" if success else "black"
    axes[1, i].set_title(f"Adv Pred: {classes[pred_label]} {mark}", color=color)
    axes[1, i].axis('off')

plt.tight_layout()
plt.show()

# 文字結果輸出
print("=== 預測結果總結 ===")
for i in range(5):
    # 從剛剛存好的 List 裡面，依序把真正的標籤和預測結果拿出來比對
    actual_label = true_labels_list[i]
    predicted_label = adv_preds_list[i]
    
    status = "Success 成功欺騙!" if actual_label != predicted_label else "Failed 失敗"
    print(f"圖片 {i+1} : 原標籤 [{classes[actual_label]:>12}] ---> 預測為 [{classes[predicted_label]:>12}] ({status})")
